  Accept: application/vnd.github+json
  Authorization: Bearer TOKEN
  X-GitHub-Api-Version: 2022-11-28

  {"event_type":"refresh-sptv"}

  */15 * * * * GITHUB_OWNER='OWNER' GITHUB_REPO='REPO' GITHUB_TOKEN='TOKEN' /bin/bash /duong-dan/external_trigger/trigger_sptv.sh >> /tmp/sptv_dispatch.log 2>&1
